## 4. Critical Implementation Rules (MUST FOLLOW)

### A. Props Usage in Templates

**WRONG:**
```vue
<template>
  <div>{{ label }}</div>
  <div v-if="loading">Loading...</div>
</template>

<script setup lang="ts">
defineProps<Props>()
</script>
```

**CORRECT:**
```vue
<template>
  <div>{{ props.label }}</div>
  <div v-if="props.loading">Loading...</div>
</template>

<script setup lang="ts">
const props = defineProps<Props>()
</script>
```

### B. Props Usage in Computed Properties

**WRONG:**
```typescript
const gridItems = computed(() => {
  return Array.from({ length: 8 }, (_, i) => metrics.value[i] || null)
})
```

**CORRECT:**
```typescript
const props = defineProps<Props>()

const gridItems = computed(() => {
  return Array.from({ length: 8 }, (_, i) => props.metrics[i] || null)
})
```

### C. Router Handling (CRITICAL)

**WRONG:**
```typescript
import { useRoute } from 'vue-router'

const route = useRoute() // This will fail if router is not configured!

const isRouteActive = (path: string): boolean => {
  return route.path.startsWith(path)
}
```

**CORRECT:**
```typescript
import { ref } from 'vue'

// Use ref-based fallback when router is not available
const currentPath = ref('/')

const isRouteActive = (path: string): boolean => {
  if (path === '/') {
    return currentPath.value === '/'
  }
  return currentPath.value.startsWith(path)
}
```

### D. defineProps and defineEmits

**WRONG:**
```typescript
defineProps<Props>()
// ... later in the file ...
const props = defineProps<Props>() // DUPLICATE!
```

**CORRECT:**
```typescript
const props = defineProps<Props>()
const emits = defineEmits<Emits>()

// Use props and emits throughout the component
```

## 6. Common Pitfalls to Avoid

1. **Duplicate defineProps()**: Only call defineProps once per component. Assign to const.
2. **Missing props prefix in templates**: Always use `props.label` not `label` in templates.
3. **Wrong computed property syntax**: Use `props.metrics[i]` not `metrics.value[i]`.
4. **Router dependency**: Do NOT import or use useRoute(). Use ref-based fallback.
5. **Unhandled errors**: Wrap router usage in try-catch or use optional chaining.
6. **Missing prop assignments**: Always assign defineProps result: `const props = defineProps<Props>()`.

## 8. Example: Correct Component Structure

```vue
<script setup lang="ts">
import { computed, ref } from 'vue'
import { SomeIcon } from 'lucide-vue-next'

interface Props {
  label: string
  value: string | number
  loading?: boolean
}

interface Emits {
  (e: 'action', payload: string): void
}

// CORRECT: Assign to const
const props = defineProps<Props>()
const emits = defineEmits<Emits>()

// CORRECT: Use ref for fallback state
const currentPath = ref('/')

// CORRECT: Use props in computed
const displayValue = computed(() => {
  return props.value || '—'
})

// CORRECT: Use props in functions
const handleClick = () => {
  emits('action', props.label)
}
</script>

<template>
  <!-- CORRECT: Use props in template -->
  <div v-if="props.loading" class="skeleton">Loading...</div>
  <div v-else>
    <p>{{ props.label }}</p>
    <p>{{ displayValue }}</p>
    <button @click="handleClick">Action</button>
  </div>
</template>
```
