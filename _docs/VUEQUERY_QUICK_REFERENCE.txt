┌─ KEY BENEFITS ────────────────────────────────────────────────────────────┐
│                                                                           │
│  1. Automatic Caching                                                    │
│     Data cached for 5 minutes by default                                 │
│     Navigate away and back - instant load!                               │
│                                                                           │
│  2. Request Deduplication                                                │
│     Multiple components requesting same data = 1 API call                │
│     Automatic sharing of cached data                                     │
│                                                                           │
│  3. Automatic Refetching                                                 │
│     After mutation, automatically refetch related data                   │
│     No manual cache invalidation needed                                   │
│                                                                           │
│  4. Background Refetching                                                │
│     Data automatically refetches in background                           │
│     Keep data fresh without user interaction                             │
│                                                                           │
│  5. Optimistic Updates                                                   │
│     Update UI immediately, sync with server after                        │
│     Better perceived performance                                         │
│                                                                           │
│  6. Built-in DevTools                                                    │
│     Visualize query state                                                │
│     Monitor cache behavior                                               │
│     Debug mutations                                                      │
│                                                                           │
└─────────────────────────────────────────────────────────────────────────┘

┌─ CODE COMPARISON ─────────────────────────────────────────────────────────┐
│                                                                           │
│  BEFORE (Pinia Store):                                                   │
│  ┌─────────────────────────────────────────────────────────────────────┐ │
│  │ export const useProductStore = defineStore('products', () => {     │ │
│  │   const products = ref([])                                         │ │
│  │   const isLoading = ref(false)                                     │ │
│  │   const error = ref(null)                                          │ │
│  │                                                                     │ │
│  │   const fetchProducts = async () => {                              │ │
│  │     isLoading.value = true                                         │ │
│  │     error.value = null                                             │ │
│  │     try {                                                          │ │
│  │       const response = await apiClient.getProducts()               │ │
│  │       if (response.success) {                                      │ │
│  │         products.value = response.data                             │ │
│  │       } else {                                                     │ │
│  │         error.value = response.error?.message                      │ │
│  │       }                                                            │ │
│  │     } catch (e) {                                                  │ │
│  │       error.value = e.message                                      │ │
│  │     } finally {                                                    │ │
│  │       isLoading.value = false                                      │ │
│  │     }                                                              │ │
│  │   }                                                                │ │
│  │                                                                     │ │
│  │   return { products, isLoading, error, fetchProducts }             │ │
│  │ })                                                                 │ │
│  │                                                                     │ │
│  │ Lines: ~80                                                         │ │
│  └─────────────────────────────────────────────────────────────────────┘ │
│                                                                           │
│  AFTER (Vue Query):                                                      │
│  ┌─────────────────────────────────────────────────────────────────────┐ │
│  │ export function useProducts() {                                    │ │
│  │   return useQuery({                                                │ │
│  │     queryKey: ['products'],                                        │ │
│  │     queryFn: () => apiClient.getProducts(),                        │ │
│  │     staleTime: 1000 * 60 * 5,                                      │ │
│  │   })                                                               │ │
│  │ }                                                                  │ │
│  │                                                                     │ │
│  │ Lines: ~10                                                         │ │
│  └─────────────────────────────────────────────────────────────────────┘ │
│                                                                           │
│  Result: 87% less code! 🎉                                               │
│                                                                           │
└─────────────────────────────────────────────────────────────────────────┘

┌─ INSTALLATION ────────────────────────────────────────────────────────────┐
│                                                                           │
│  1. Install package                                                      │
│     npm install @tanstack/vue-query                                      │
│                                                                           │
│  2. Create query client (src/lib/queryClient.ts)                         │
│     import { QueryClient } from '@tanstack/vue-query'                    │
│                                                                           │
│     export const queryClient = new QueryClient({                         │
│       defaultOptions: {                                                  │
│         queries: {                                                       │
│           staleTime: 1000 * 60 * 5,                                      │
│           gcTime: 1000 * 60 * 10,                                        │
│         },                                                               │
│       },                                                                 │
│     })                                                                   │
│                                                                           │
│  3. Setup plugin (src/main.ts)                                           │
│     import { VueQueryPlugin } from '@tanstack/vue-query'                 │
│     import { queryClient } from './lib/queryClient'                      │
│                                                                           │
│     app.use(VueQueryPlugin, { queryClient })                             │
│                                                                           │
│  Done! ✅                                                                │
│                                                                           │
└─────────────────────────────────────────────────────────────────────────┘

┌─ BASIC USAGE ─────────────────────────────────────────────────────────────┐
│                                                                           │
│  Fetching Data:                                                          │
│  ┌─────────────────────────────────────────────────────────────────────┐ │
│  │ const { data, isLoading, error } = useQuery({                      │ │
│  │   queryKey: ['products'],                                          │ │
│  │   queryFn: () => apiClient.getProducts(),                          │ │
│  │ })                                                                 │ │
│  └───────────────────────────────────────���─────────────────────────────┘ │
│                                                                           │
│  Mutating Data:                                                          │
│  ┌─────────────────────────────────────────────────────────────────────┐ │
│  │ const { mutate } = useMutation({                                   │ │
│  │   mutationFn: (data) => apiClient.createProduct(data),             │ │
│  │   onSuccess: () => {                                               │ │
│  │     queryClient.invalidateQueries({ queryKey: ['products'] })      │ │
│  │   },                                                               │ │
│  │ })                                                                 │ │
│  └─────────────────────────────────────────────────────────────────────┘ │
│                                                                           │
│  In Components:                                                          │
│  ┌─────────────────────────────────────────────────────────────────────┐ │
│  │ <script setup>                                                     │ │
│  │ const { data: products, isLoading, error } = useProducts()         │ │
│  │ </script>                                                          │ │
│  │                                                                     │ │
│  │ <template>                                                         │ │
│  │   <div v-if="isLoading">Loading...</div>                           │ │
│  │   <div v-else-if="error">{{ error.message }}</div>                 │ │
│  │   <div v-else>                                                     │ │
│  │     <div v-for="product in products" :key="product.id">            │ │
│  │       {{ product.name }}                                           │ │
│  │     </div>                                                         │ │
│  │   </div>                                                           │ │
│  │ </template>                                                        │ │
│  └─────────────────────────────────────────────────────────────────────┘ │

┌─ PERFORMANCE IMPACT ──────────────────────────────────────────────────────┐
│                                                                           │
│  API Calls Reduction:                                                    │
│                                                                           │
│  Without Vue Query:                                                      │
│    Page 1 → Fetch (1s)                                                   │
│    Page 2 → Fetch (1s)                                                   │
│    Back to Page 1 → Fetch (1s) ❌ UNNECESSARY                            │
│    Total: 3 API calls                                                    │
│                                                                           │
│  With Vue Query:                                                         │
│    Page 1 → Fetch (1s)                                                   │
│    Page 2 → Fetch (1s)                                                   │
│    Back to Page 1 → Cache (instant) ✅ FAST                              │
│    Total: 2 API calls                                                    │
│                                                                           │
│  Improvement: 33% fewer API calls                                        │
│  (Can be 50-70% with proper configuration)                               │
